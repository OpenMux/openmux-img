
# OpenMux serial console (appended to the chroot's /boot/firmware/config.txt
# by the layer's late preimage hook preimage20-config.txt, which runs after
# every layer's customize phase work). It lands AFTER the rpi-boot-firmware
# template, i.e. appended after ALL [ ] sections, so it applies to [all].
#
# BCM2712 (Pi 5 / CM5) console routing: without these, the active kernel
# console ends up on a DEBUG uart - on the Pi 5 that is ttyAMA0, which is
# NOT available on the CM5 pin header. Enable uart0 (the PL011 on the
# 40-pin header) and route the console to it, so console=serial0 lands on
# /dev/ttyAMA0 on the CM5.
dtparam=uart0=on
dtparam=uart0_console=on
# (The template's [cm5] section carries no enable_uart - [all] only has
# uart_2ndstage=1 - so it is set here for the pre-driver firmware window;
# the template's other sections may already set it, which is harmless.)
enable_uart=1
